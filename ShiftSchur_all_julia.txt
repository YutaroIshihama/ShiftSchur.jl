"""
    auxiliary_matrix(M)

Frequency-independent enlarged matrix [A B; B† D]; (auxiliary_matrix(M) - zI)
is the full (unreduced) enlarged system whose Schur complement is S(z).
"""
auxiliary_matrix(M::SchurModel) = [Matrix(M.A) Matrix(M.B); Matrix(M.B') Matrix(M.D)]

"""
    solve_full_upper(M, z, b)

Upper block x of the solution of
`[A-zI B; B† D-zI] [x; y] = [b; 0]`.
"""
function solve_full_upper(M::SchurModel, z::Number, b)
    n = size(M.A, 1)
    H = auxiliary_matrix(M)
    rhs = [b; zeros(eltype(H), size(M.D, 1), size(b)[2:end]...)]
    sol = (H - z * I) \ rhs
    sol[1:n, ntuple(_ -> :, ndims(b) - 1)...]
end




===== FILE: src\operators\schur_operator.jl =====

"""
    apply_schur(M, z, v)

S(z) v = (A - zI) v - Σ(z) v, with Σ(z) = B (D - zI)⁻¹ B†.
"""
apply_schur(M::SchurModel, z::Number, v) =
    M.A * v - z * v - apply_self_energy(M, z, v)

"""
    schur_matrix(M, z)

Dense S(z) = A - zI - B (D - zI)⁻¹ B†.
"""
schur_matrix(M::SchurModel, z::Number) = Matrix(M.A) - z * I - self_energy(M, z)

"""
    solve_schur(M, z, b)

Direct solve of S(z) x = b (exact Schur system, no approximation).
"""
solve_schur(M::SchurModel, z::Number, b) = schur_matrix(M, z) \ b




===== FILE: src\operators\schur_self_energy.jl =====

"""
    SchurModel(A, D, B)

Hermitian `A` (n×n, retained space), Hermitian `D` (m×m, eliminated space)
and coupling `B` (n×m). Convention: `z = ω + iη`, shifts enter as `X - zI`.
"""
struct SchurModel{TA,TD,TB}
    A::TA
    D::TD
    B::TB
end

"""
    apply_self_energy(M, z, v)

Σ(z) v = B (D - zI)⁻¹ B† v  (matrix-compatible; one D-solve per call).
"""
apply_self_energy(M::SchurModel, z::Number, v) =
    M.B * ((M.D - z * I) \ (M.B' * v))

"""
    self_energy(M, z)

Dense Σ(z) = B (D - zI)⁻¹ B†.
"""
self_energy(M::SchurModel, z::Number) = M.B * ((M.D - z * I) \ Matrix(M.B'))




===== FILE: src\reduction\adaptive_poles.jl =====

"""
    greedy_rational_shifts(M, ωgrid, η, b; s_max=12, rtol=1e-10, init_shifts=ComplexF64[im * η])

Greedily select rational Krylov shifts for the self-energy action on `b`.
Starting from `init_shifts`, repeatedly build a basis from `M.D` and `M.B'`,
then add the shift `ω + im*η` at the grid point with largest relative error
`norm((Σ(z) - Σₖ(z)) * b) / norm(Σ(z) * b)`, where `z = ω + im*η`,
`Σ(z) = self_energy(M, z)`, and `Σₖ(z) = reduced_self_energy(RM, z)`.
The process stops when the maximum error is below `1e-8` or after `s_max`
shifts have been added. Repeated candidate shifts are slightly perturbed.

Returns the selected shifts and a vector of per-iteration named tuples with
`max_err` and basis dimension `k`.
"""
function greedy_rational_shifts(M::SchurModel, ωgrid, η::Real, b;
                                s_max::Integer = 12, rtol::Real = 1e-10,
                                init_shifts = ComplexF64[im * η])
    s_max >= 0 || throw(ArgumentError("s_max must be nonnegative"))
    frequencies = collect(ωgrid)
    isempty(frequencies) && throw(ArgumentError("ωgrid must not be empty"))

    shifts = ComplexF64.(collect(init_shifts))
    history = NamedTuple[]
    for added in 0:s_max
        Q = rational_krylov_basis(M.D, M.B', shifts; rtol = rtol)
        RM = ReducedSchurModel(M, Q)
        errors = Float64[]
        for ω in frequencies
            z = ω + im * η
            exact = self_energy(M, z) * b
            approx = reduced_self_energy(RM, z) * b
            denominator = norm(exact)
            err = denominator == 0 ? (norm(exact - approx) == 0 ? 0.0 : Inf) :
                                     norm(exact - approx) / denominator
            push!(errors, err)
        end

        max_err, worst = findmax(errors)
        push!(history, (max_err = max_err, k = size(Q, 2)))
        (max_err < 1e-8 || added == s_max) && break

        candidate = ComplexF64(frequencies[worst] + im * η)
        while any(abs(candidate - σ) <= 100eps(Float64) * max(1.0, abs(candidate), abs(σ))
                  for σ in shifts)
            candidate += sqrt(eps(Float64)) * max(1.0, abs(candidate))
        end
        push!(shifts, candidate)
    end
    return shifts, history
end




===== FILE: src\reduction\diagnostics.jl =====

"""
    rational_compression_diagnostics(M, ωgrid, η, b, shifts_list; kind = :rational, rtol = 1e-10)

For each shift set `σ` in `shifts_list` (for `kind = :polynomial` the entries
are integers = number of polynomial steps), build the basis from the starting
block `M.B'`, reduce, and evaluate on `z = ω + iη`, `ω ∈ ωgrid`:

* `Σerr_max/mean` : ‖Σ(z)-Σ_k(z)‖_F/‖Σ(z)‖_F  (ROM approximation error)
* `xerr_max`      : ‖x_exact-x_ROM‖/‖x_exact‖, x_exact = `solve_schur`, x_ROM = `solve_reduced_schur`
* `lin_err_max`   : ‖x_aux-x_ROM‖/‖x_ROM‖ with x_aux = `solve_full_reduced_upper`
                    (algebraic linearization identity error, NOT approximation error)

Returns a vector of NamedTuples `(kind, η, nshifts, k, Σerr_max, Σerr_mean, xerr_max, lin_err_max)`.
"""
function rational_compression_diagnostics(M::SchurModel, ωgrid, η::Real, b, shifts_list;
                                          kind::Symbol = :rational, rtol::Real = 1e-10)
    rows = NamedTuple[]
    for sh in shifts_list
        Q = kind === :rational ? rational_krylov_basis(M.D, M.B', sh; rtol = rtol) :
                                 polynomial_krylov_basis(M.D, M.B', sh; rtol = rtol)
        RM = ReducedSchurModel(M, Q)
        es = Float64[]; ex = Float64[]; el = Float64[]
        for ω in ωgrid
            z = ω + im * η
            Σ = self_energy(M, z)
            push!(es, norm(Σ - reduced_self_energy(RM, z)) / norm(Σ))
            x = solve_schur(M, z, b)
            xr = solve_reduced_schur(RM, z, b)
            push!(ex, norm(x - xr) / norm(x))
            push!(el, norm(solve_full_reduced_upper(RM, z, b) - xr) / norm(xr))
        end
        push!(rows, (kind = kind, η = η, nshifts = sh isa Integer ? sh : length(sh),
                     k = size(Q, 2), Σerr_max = maximum(es), Σerr_mean = sum(es) / length(es),
                     xerr_max = maximum(ex), lin_err_max = maximum(el)))
    end
    return rows
end

"""
    print_diagnostics(rows; io = stdout)

Print a table of rows returned by [`rational_compression_diagnostics`](@ref).
"""
function print_diagnostics(rows; io::IO = stdout)
    println(io, "kind        eta      s    k    max Σerr    mean Σerr   max xerr    max lin(aux-ROM)")
    for r in rows
        println(io, rpad(string(r.kind), 11), " ", rpad(r.η, 8), " ", rpad(r.nshifts, 4), " ", rpad(r.k, 4), " ",
                join((rpad(string(round(v; sigdigits = 3)), 11) for v in (r.Σerr_max, r.Σerr_mean, r.xerr_max, r.lin_err_max)), " "))
    end
end




===== FILE: src\reduction\projection.jl =====

"""
    ReducedSchurModel(A, T, Bred, Q)
    ReducedSchurModel(M::SchurModel, Q)

Galerkin-projected model with `T = Q†DQ` (k×k Hermitian) and `Bred = BQ`
(n×k). Convention: `z = ω + iη`, shifts enter as `X - zI`. The exact
[`SchurModel`](@ref) is left untouched.
"""
struct ReducedSchurModel{TA,TT,TB,TQ}
    A::TA
    T::TT
    Bred::TB
    Q::TQ
end

function ReducedSchurModel(M::SchurModel, Q::AbstractMatrix)
    ReducedSchurModel(M.A, Q' * (M.D * Q), M.B * Q, Q)
end

"""
    reduced_self_energy(RM, z)

Dense Σ_k(z) = Bred (T - zI)⁻¹ Bred†.
"""
reduced_self_energy(RM::ReducedSchurModel, z::Number) =
    RM.Bred * ((Matrix(RM.T) - z * I) \ Matrix(RM.Bred'))

"""
    apply_reduced_self_energy(RM, z, v)

Σ_k(z) v = Bred (T - zI)⁻¹ Bred† v.
"""
apply_reduced_self_energy(RM::ReducedSchurModel, z::Number, v) =
    RM.Bred * ((Matrix(RM.T) - z * I) \ (RM.Bred' * v))

"""
    reduced_schur_matrix(RM, z)

Dense S_k(z) = A - zI - Σ_k(z).
"""
reduced_schur_matrix(RM::ReducedSchurModel, z::Number) =
    Matrix(RM.A) - z * I - reduced_self_energy(RM, z)

"""
    solve_reduced_schur(RM, z, b)

Direct solve of S_k(z) x = b (approximate Schur system from the reduced model).
"""
solve_reduced_schur(RM::ReducedSchurModel, z::Number, b) = reduced_schur_matrix(RM, z) \ b

"""
    reduced_auxiliary_matrix(RM)

H_aux,k = [A Bred; Bred† T]; (H_aux,k - zI) is the enlarged system whose
Schur complement is S_k(z).
"""
reduced_auxiliary_matrix(RM::ReducedSchurModel) =
    [Matrix(RM.A) Matrix(RM.Bred); Matrix(RM.Bred') Matrix(RM.T)]

"""
    solve_full_reduced_upper(RM, z, b)

Upper block x_aux of the solution of `(H_aux,k - zI)[x; y] = [b; 0]`.
Algebraically equal to `solve_reduced_schur(RM, z, b)` (linearization
identity, independent of how well the ROM approximates the exact model).
"""
function solve_full_reduced_upper(RM::ReducedSchurModel, z::Number, b)
    n = size(RM.A, 1)
    H = reduced_auxiliary_matrix(RM)
    rhs = [b; zeros(eltype(H), size(RM.T, 1), size(b)[2:end]...)]
    sol = (H - z * I) \ rhs
    sol[1:n, ntuple(_ -> :, ndims(b) - 1)...]
end




===== FILE: src\reduction\rational_krylov.jl =====

# Orthonormal basis of range(V) via SVD, discarding directions with
# singular value <= rtol * ||V||_F. Returns an m×r matrix (r may be 0).
function _orth_range(V::AbstractMatrix, rtol::Real)
    nrm = norm(V)
    (nrm == 0 || size(V, 2) == 0) && return zeros(eltype(V), size(V, 1), 0)
    F = svd(V)
    r = count(>(rtol * nrm), F.S)
    return F.U[:, 1:r]
end

# Orthonormal basis of range(V) ⟂ range(Q): two block Gram-Schmidt passes,
# rank-revealing SVD, then one more cleanup pass.
function _orth_against(Q::AbstractMatrix, V::AbstractMatrix, rtol::Real)
    nrm = norm(V)
    nrm == 0 && return zeros(eltype(V), size(V, 1), 0)
    R = V
    for _ in 1:2
        R = R - Q * (Q' * R)
    end
    F = svd(R)
    r = count(>(rtol * nrm), F.S)
    r == 0 && return zeros(eltype(V), size(V, 1), 0)
    U = F.U[:, 1:r]
    U = U - Q * (Q' * U)
    return Matrix(qr(U).Q)[:, 1:r]
end

function _krylov_basis(apply, W::AbstractMatrix, nsteps::Integer, rtol::Real)
    T = promote_type(eltype(W), ComplexF64)
    Wc = Matrix{T}(W)
    Vk = _orth_range(Wc, rtol)
    Q = Vk
    for j in 1:nsteps
        size(Vk, 2) == 0 && break
        # continue from the full (undeflated) orthonormal block, so the space
        # is exactly the stated Krylov space
        Vk = _orth_range(apply(j, Vk), rtol)
        Q = hcat(Q, _orth_against(Q, Vk, rtol))
    end
    return Q
end

"""
    rational_krylov_basis(D, W, shifts; rtol = 1e-10)

Orthonormal basis `Q` (m×k, `Q†Q ≈ I`) of the block rational Krylov space

    K = span{ W, (D-σ₁I)⁻¹W, (D-σ₂I)⁻¹(D-σ₁I)⁻¹W, …, ∏_{j=1}^{s}(D-σⱼI)⁻¹ W }

for Hermitian `D` (m×m), starting block `W` (m×p, e.g. `W = B†`) and the
explicitly supplied `shifts = [σ₁,…,σₛ]` (may be complex, must avoid the
spectrum of `D`). Each block is the previous block (not only its new part)
multiplied by `(D-σⱼI)⁻¹`. Equivalently
`K = { r(D) w : w ∈ range(W), r = p/∏(z-σⱼ), deg p ≤ s }` — a rational Krylov
space with prescribed poles σⱼ (single sequential pole sequence, no adaptivity).

Rank-revealing: each block is orthonormalized by SVD, directions with singular
value ≤ `rtol`·‖block‖_F are dropped, and orthogonalization against the
existing basis is done with repeated Gram-Schmidt. Hence `k ≤ p(s+1)`; a zero
or rank-deficient `W` yields fewer (possibly zero) columns. Dense proof of
concept (one LU of `D-σⱼI` per shift).
"""
function rational_krylov_basis(D::AbstractMatrix, W::AbstractMatrix, shifts; rtol::Real = 1e-10)
    σ = collect(shifts)
    Dd = Matrix(D)
    facts = [lu(Dd - s * I) for s in σ]
    return _krylov_basis((j, V) -> facts[j] \ V, W, length(σ), rtol)
end

"""
    polynomial_krylov_basis(D, W, nsteps; rtol = 1e-10)

Orthonormal basis of the block polynomial Krylov space
`span{W, DW, D²W, …, D^nsteps W}`, built with the same rank-revealing
orthogonalization as [`rational_krylov_basis`](@ref). Used as a comparison baseline.
"""
function polynomial_krylov_basis(D::AbstractMatrix, W::AbstractMatrix, nsteps::Integer; rtol::Real = 1e-10)
    Dd = Matrix(D)
    return _krylov_basis((j, V) -> Dd * V, W, nsteps, rtol)
end

using LinearAlgebra, Random

# Synthetic model: D (m=150) has a gapped bulk spectrum in [-3,-1.3] ∪ [1.3,3]
# plus 4 discrete "resonance" levels inside the frequency interval [-1,1];
# B (n=3 × m) couples strongly to the resonances and weakly to the bulk.
# Random unitary eigenbasis, fixed seed => deterministic.
function synthetic_rational_model(; m = 150, n = 3, seed = 20240601)
    rng = MersenneTwister(seed)
    nb = m - 4
    bulk = [range(-3.0, -1.3; length = nb ÷ 2); range(1.3, 3.0; length = nb - nb ÷ 2)]
    λ = sort([bulk; -0.6; -0.1; 0.35; 0.8])
    U = Matrix(qr(randn(rng, ComplexF64, m, m)).Q)
    D = U * Diagonal(λ) * U'
    D = (D + D') / 2
    A = (X = randn(rng, ComplexF64, n, n); (X + X') / 2)
    w = [abs(x) < 1.1 ? 0.5 : 3 / sqrt(m) for x in λ]
    B = randn(rng, ComplexF64, n, m) * Diagonal(w) * U'
    b = randn(rng, ComplexF64, n)
    return SchurModel(A, D, B), b
end

@testset "Rational compression of Σ(z)" begin
    M, b = synthetic_rational_model()
    m = size(M.D, 1)
    ωmin, ωmax = -1.0, 1.0
    ωgrid = range(ωmin, ωmax; length = 41)
    ηs = (0.2, 0.1, 0.05, 0.02)
    shift_sets(s, η) = collect(range(ωmin, ωmax; length = s) .+ im * 0.1)
    svals = (2, 4, 8, 12, 16)

    @testset "basis properties" begin
        for s in svals
            Q = rational_krylov_basis(M.D, M.B', shift_sets(s, 0.1))
            RM = ReducedSchurModel(M, Q)
            @test size(Q, 2) <= 3 * (s + 1)
            @test size(Q, 2) >= 3 * (s + 1) - 3   # at most mild deflation
            @test norm(Q' * Q - I) < 1e-12
            @test norm(RM.T - RM.T') < 1e-12 * norm(RM.T)
            @test size(RM.T) == (size(Q, 2), size(Q, 2))
            @test size(RM.Bred) == (3, size(Q, 2))
            @test RM.Bred ≈ M.B * Q
        end
        # starting block contained in the space
        Q = rational_krylov_basis(M.D, M.B', shift_sets(3, 0.1))
        @test norm(M.B' - Q * (Q' * M.B')) < 1e-12 * norm(M.B')
    end

    @testset "rank-deficient / zero starting block" begin
        sh = shift_sets(3, 0.1)
        W = M.B'[:, [1, 1, 2]]                       # repeated column
        Q = rational_krylov_basis(M.D, W, sh)
        @test size(Q, 2) == 2 * 4
        @test norm(Q' * Q - I) < 1e-12
        Q0 = rational_krylov_basis(M.D, zeros(ComplexF64, m, 2), sh)
        @test size(Q0) == (m, 0)
        RM0 = ReducedSchurModel(M, Q0)
        @test norm(reduced_self_energy(RM0, 0.3 + 0.1im)) == 0
        @test solve_reduced_schur(RM0, 0.3 + 0.1im, b) ≈ (M.A - (0.3 + 0.1im) * I) \ b
        Q1 = rational_krylov_basis(M.D, M.B'[:, 1:1], sh)   # single vector start
        @test size(Q1, 2) == 4
        @test norm(Q1' * Q1 - I) < 1e-12
    end

    @testset "reduced linearization identity (x_aux == x_ROM, algebraic)" begin
        for s in (2, 6), z in (0.3 + 0.05im, -0.7 + 0.2im, 5.0 + 1.0im)
            RM = ReducedSchurModel(M, rational_krylov_basis(M.D, M.B', shift_sets(s, 0.1)))
            xr = solve_reduced_schur(RM, z, b)
            xa = solve_full_reduced_upper(RM, z, b)
            @test norm(xa - xr) / norm(xr) < 1e-11
            Haux = reduced_auxiliary_matrix(RM)
            @test norm(Haux - Haux') < 1e-12 * norm(Haux)
            v = b .+ 1
            @test apply_reduced_self_energy(RM, z, v) ≈ reduced_self_energy(RM, z) * v
        end
    end

    @testset "ROM approximation error over whole grid (x_ROM vs x_exact)" begin
        all_rows = NamedTuple[]
        for η in ηs
            rows = rational_compression_diagnostics(M, ωgrid, η, b, [shift_sets(s, η) for s in svals])
            append!(all_rows, rows)
            @test all(r.lin_err_max < 1e-9 for r in rows)
        end
        println("\n=== Rational Krylov (shifts ω_j + 0.1i), errors vs eta ===")
        print_diagnostics(all_rows)
        poly = NamedTuple[]
        for η in ηs
            append!(poly, rational_compression_diagnostics(M, ωgrid, η, b, collect(svals); kind = :polynomial))
        end
        println("\n=== Polynomial Krylov baseline ===")
        print_diagnostics(poly)

        r(η, s) = only(filter(x -> x.η == η && x.nshifts == s, all_rows))
        pr(kind, η, sv) = only(filter(x -> x.η == η && x.nshifts == sv, kind === :rational ? all_rows : poly))
        for η in ηs
            @test pr(:rational, η, 8).Σerr_max < 1e-3
            @test pr(:rational, η, 8).xerr_max < 1e-3
            @test pr(:rational, η, 16).Σerr_max < 1e-7
            @test pr(:rational, η, 16).xerr_max < 1e-7
            @test pr(:rational, η, 8).Σerr_max < pr(:rational, η, 4).Σerr_max / 100
            # rational beats polynomial Krylov at the same number of blocks
            @test pr(:rational, η, 8).Σerr_max < pr(:polynomial, η, 8).Σerr_max / 100
        end
        # smaller eta (closer to the real-axis poles) is harder at fixed small dimension
        @test pr(:rational, 0.02, 4).Σerr_max > pr(:rational, 0.2, 4).Σerr_max
    end
end

using LinearAlgebra, Random

@testset "Schur identity (random Hermitian)" begin
    rng = MersenneTwister(1234)
    n, m = 6, 14
    hermitian(k) = (X = randn(rng, ComplexF64, k, k); (X + X') / 2)
    A, D = hermitian(n), hermitian(m)
    B = randn(rng, ComplexF64, n, m) / 4
    b = randn(rng, ComplexF64, n)
    M = SchurModel(A, D, B)
    H = auxiliary_matrix(M)
    @test ishermitian(H)
    for z in (0.3 + 1e-2im, -1.5 + 0.1im, 2.0 + 1.0im, 40.0 + 0.5im, 0.0 + 1.0im)
        x = solve_schur(M, z, b)

        @test norm(apply_schur(M, z, x) - b) / norm(b) < 1e-11

        @test apply_self_energy(M, z, v) ≈ self_energy(M, z) * v
    end
    # real z outside the spectrum
    zr = maximum(eigvals(Hermitian(H))) + 1.0
    @test norm(solve_schur(M, zr, b) - solve_full_upper(M, zr, b)) < 1e-11 * norm(solve_schur(M, zr, b))
